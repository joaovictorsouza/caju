# Comandos de compilação cruzada e execução do Observation no Galaxy S22+

## 1. Configuração de Variáveis de Ambiente
export ANDROID_NDK_HOME=$HOME/Library/Android/sdk/ndk/30.0.15729638
export ANDROID_NDK_ROOT=$HOME/Library/Android/sdk/ndk/30.0.15729638
export ANDROID_HOME=$HOME/Library/Android/sdk

## 2. Compilação dos Objetos Swift (sem autolink do SPM no Mac)
cd /tmp/obs-android
swift build -c release --swift-sdk aarch64-unknown-linux-android29 --target obs-android || true

## 3. Localização de Runtime e Ferramentas do NDK
SDK_ROOT="$(ls -d "$HOME"/Library/org.swift.swiftpm/swift-sdks/*android*.artifactbundle/swift-android | head -n 1)"
RUNTIME="$SDK_ROOT/swift-resources/usr/lib/swift-aarch64/android"
LLVM="$(ls -d "$ANDROID_NDK_HOME"/toolchains/llvm/prebuilt/*)"
OBJECTS="$(find .build/out/Intermediates.noindex -path "*Release-android-aarch64*" -name '*.o')"
mkdir -p /tmp/obs-android/out

## 4. Link Manual com Clang/LLD do NDK
"$LLVM/bin/aarch64-linux-android29-clang" -fuse-ld=lld \
  -Wl,-z,max-page-size=16384 \
  -Wl,-rpath,'$ORIGIN' \
  -o /tmp/obs-android/out/obs-android \
  $OBJECTS \
  "$RUNTIME/aarch64/swiftrt.o" \
  -L"$RUNTIME" \
  -lswiftCore \
  -lswift_Concurrency \
  -lswift_StringProcessing \
  -lswiftSynchronization \
  -lswiftObservation \
  -lswiftAndroid \
  -lc++_shared \
  -ldl \
  -lm

## 5. Cópia do libc++_shared.so e Dependências Transitivas do Swift Runtime
cp "$LLVM/sysroot/usr/lib/aarch64-linux-android/libc++_shared.so" /tmp/obs-android/out/
# Resolução transitiva via llvm-readelf de DT_NEEDED (libdispatch, libBlocksRuntime, libswift_Builtin_float, libswift_RegexParser)

## 6. Envio e Execução no Galaxy S22+ (SM-S906E) via ADB
adb -s RQCT202D4FM shell rm -rf /data/local/tmp/obs
adb -s RQCT202D4FM shell mkdir -p /data/local/tmp/obs
adb -s RQCT202D4FM push /tmp/obs-android/out/* /data/local/tmp/obs/
adb -s RQCT202D4FM shell chmod +x /data/local/tmp/obs/obs-android
adb -s RQCT202D4FM shell "cd /data/local/tmp/obs && LD_LIBRARY_PATH=. ./obs-android"
